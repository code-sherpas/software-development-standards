Increment 3 of PROJ-402 (“Search invoices by asking”).

**Increment:** the invoices list gets a search box where an accountant types what they want — “unpaid from last quarter over 500 EUR”, “everything ACME sent us in March” — and the list filters to it. The text goes to the LLM with one tool, `filter_invoices`, whose arguments become the list's filter; the filter chips above the list show what was understood so it can be corrected by hand. Increments 1 and 2 (manual filter chips, the list endpoint taking filters) are in production.

**Acceptance:**
- “unpaid from last quarter over 500 EUR” on 2 November 2026 shows unpaid invoices issued from 1 July to 30 September 2026 with a total above 500 EUR.
- A request the tool can't express (“the ones I liked”) leaves the filters untouched and says it didn't understand.

+import Anthropic from '@anthropic-ai/sdk'
+import { ResultAsync, errAsync, okAsync } from 'neverthrow'
+import { InvoiceFilter } from '../invoice-filter'
+
+const client = new Anthropic()
+
+const filterInvoicesTool = {
+  name: 'filter_invoices',
+  description: 'Filter the invoice list',
+  input_schema: {
+    type: 'object',
+    properties: {
+      status: { type: 'string', enum: ['draft', 'issued', 'paid', 'overdue'] },
+      issuedFrom: { type: 'string', description: 'date' },
+      issuedTo: { type: 'string', description: 'date' },
+      minTotal: { type: 'number' },
+      currency: { type: 'string' },
+      supplierName: { type: 'string' },
+    },
+  },
+} as const
+
+type InterpretError = { kind: 'not-understood' } | { kind: 'provider-failed'; cause: unknown }
+
+export function interpretInvoiceSearch(text: string): ResultAsync<InvoiceFilter, InterpretError> {
+  return ResultAsync.fromPromise(
+    client.messages.create({
+      model: 'claude-sonnet-5-5',
+      max_tokens: 512,
+      tools: [filterInvoicesTool],
+      tool_choice: { type: 'auto' },
+      messages: [{ role: 'user', content: text }],
+    }),
+    (cause): InterpretError => ({ kind: 'provider-failed', cause }),
+  ).andThen(function toFilter(response) {
+    const call = response.content.find((block) => block.type === 'tool_use')
+    if (!call) return errAsync<InvoiceFilter, InterpretError>({ kind: 'not-understood' })
+    const input = call.input as Record<string, string | number | undefined>
+    return okAsync({
+      status: input.status as InvoiceFilter['status'],
+      issuedFrom: input.issuedFrom ? new Date(input.issuedFrom as string) : undefined,
+      issuedTo: input.issuedTo ? new Date(input.issuedTo as string) : undefined,
+      minTotal: input.minTotal as number | undefined,
+      currency: input.currency as string | undefined,
+      supplierName: input.supplierName as string | undefined,
+    })
+  })
+}
