+export class InvoiceCannotBePaidError extends Error {
+  constructor(readonly invoiceId: InvoiceId, readonly status: InvoiceStatus) {
+    super(`Invoice ${invoiceId} is ${status} and cannot be marked as paid`)
+  }
+}
+
+export function newInvoiceId(): InvoiceId {
+  return randomUUID() as InvoiceId
+}
+
+export function newInvoiceLineId(): InvoiceLineId {
+  return randomUUID() as InvoiceLineId
+}
+
+export function draftInvoice(
+  customerId: CustomerId,
+  lines: ReadonlyArray<Omit<InvoiceLine, 'id'>>,
+): Invoice {
+  return {
+    id: newInvoiceId(),
+    customerId,
+    status: 'draft',
+    lines: lines.map(function withId(line) {
+      return { ...line, id: newInvoiceLineId() }
+    }),
+  }
+}
+
+export function issueInvoice(invoice: Invoice): Invoice {
+  return { ...invoice, status: 'issued' }
+}
+
+export function markAsPaid(
+  invoice: Invoice,
+  paidOn: Temporal.PlainDate,
+): Result<Invoice, InvoiceCannotBePaidError> {
+  if (invoice.status !== 'issued') {
+    return err(new InvoiceCannotBePaidError(invoice.id, invoice.status))
+  }
+  return ok({ ...invoice, status: 'paid', paidOn })
+}

+export class InvoiceTotalsService {
+  lineTotalCents(line: InvoiceLine): number {
+    return line.quantity * line.unitPriceCents
+  }
+
+  totalCents(invoice: Invoice): number {
+    let total = 0
+    for (const line of invoice.lines) {
+      total += this.lineTotalCents(line)
+    }

+  }
