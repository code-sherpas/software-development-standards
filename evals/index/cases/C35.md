The project uses its own design system, `@hearth/components`, which has a `ProgressBar`, a `Banner` and a `Button`. The diff under review:

```diff
@@ -0,0 +1,40 @@
+<script setup lang="ts">
+import { onMounted, onUnmounted, ref } from "vue";
+import { fetchExport } from "../api/exports";
+
+const props = defineProps<{ exportId: string }>();
+const status = ref<string>("queued");
+const downloadUrl = ref<string | null>(null);
+let timer: number | undefined;
+
+async function refresh() {
+  const exp = await fetchExport(props.exportId);
+  status.value = exp.status;
+  downloadUrl.value = exp.downloadUrl;
+}
+
+onMounted(() => {
+  refresh();
+  timer = window.setInterval(refresh, 3000);
+});
+onUnmounted(() => window.clearInterval(timer));
+</script>
+
+<template>
+  <div class="export-status">
+    <div v-if="status !== 'ready'" class="spinner-wrap">
+      <div class="spinner" />
+      <span>Preparing your export…</span>
+    </div>
+    <a v-else :href="downloadUrl ?? '#'" class="download-link">Download</a>
+  </div>
+</template>
+
+<style scoped>
+.spinner { width: 18px; height: 18px; border: 2px solid #c7c7c7; border-top-color: #3b6ef5; border-radius: 50%; animation: spin 1s linear infinite; }
+.download-link { color: #3b6ef5; font-weight: 600; }
+@keyframes spin { to { transform: rotate(360deg); } }
+</style>
```
