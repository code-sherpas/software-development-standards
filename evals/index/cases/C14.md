Task PROJ-430 «Front desk books appointments», increment 2 of 4: **create an appointment.** Increment 1 (clinician weekly schedules) is verified in production. The booking form arrives in increment 3; this one ships the command and its server action.

Acceptance:

- A signed-in staff member with the `appointments:write` permission on the clinic can book a patient with a clinician of that clinic, for a start time and a duration.
- The appointment must fall inside the clinician's working hours for that day, and must not overlap any other appointment of the same clinician.
- Two front desks booking the same slot at the same moment never both succeed.
- The response is the new appointment's id.

+import { errAsync, okAsync, ResultAsync } from 'neverthrow'
+import { Appointment } from './appointment'
+import { ClinicianSchedule } from './clinician-schedule'
+
+export class SlotNotBookableError extends Error {
+  constructor(readonly reason: 'outside-working-hours' | 'overlaps-appointment') {
+    super(`Slot not bookable: ${reason}`)
+  }
+}
+
+export function ensureSlotIsBookable(
+  schedule: ClinicianSchedule,
+  existing: ReadonlyArray<Appointment>,
+  startsAt: Temporal.Instant,
+  endsAt: Temporal.Instant,
+): ResultAsync<void, SlotNotBookableError> {
+  const day = startsAt.toZonedDateTimeISO(schedule.timeZone)
+  const hours = schedule.workingHoursOn(day.dayOfWeek)
+  const start = day.toPlainTime()
+  const end = endsAt.toZonedDateTimeISO(schedule.timeZone).toPlainTime()
+  if (!hours || Temporal.PlainTime.compare(start, hours.from) < 0 || Temporal.PlainTime.compare(end, hours.to) > 0) {
+    return errAsync(new SlotNotBookableError('outside-working-hours'))
+  }
+  const overlaps = existing.some(function overlapsSlot(appointment) {
+    return Temporal.Instant.compare(appointment.startsAt, endsAt) < 0 &&
+      Temporal.Instant.compare(startsAt, appointment.endsAt) < 0
+  })
+  if (overlaps) {
+    return errAsync(new SlotNotBookableError('overlaps-appointment'))
+  }
+  return okAsync(undefined)
+}
