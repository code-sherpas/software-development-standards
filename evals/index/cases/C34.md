+import { useEffect, useState } from "react";
+import { listArchivedProjects } from "../server/projects.actions";
+
+type ArchivedProject = { id: string; name: string; archivedAt: string };
+
+export function ArchivedProjectsTab() {
+  const [projects, setProjects] = useState<ArchivedProject[]>([]);
+  const [query, setQuery] = useState("");
+
+  useEffect(() => {
+    listArchivedProjects().then((result) => {
+      if (result.ok) setProjects(result.value);
+    });
+  }, []);
+
+  const shown = projects.filter((p) => p.name.toLowerCase().includes(query.toLowerCase()));
+
+  return (
+    <div>
+      <input placeholder="Search" value={query} onChange={(e) => setQuery(e.target.value)} />
+      {shown.length === 0 ? (
+        <p>No projects</p>
+      ) : (
+        <ul>
+          {shown.map((p) => (
+            <li key={p.id}>
+              {p.name} — archived {new Date(p.archivedAt).toLocaleDateString()}
+            </li>
+          ))}
+        </ul>
+      )}
+    </div>
+  );
+}
