Task PROJ-446 «Organizations move paid seats between their teams», increment 1 of 2: **transfer seats.** Increment 2 adds the screen; this one ships the command.

Acceptance:

- The owner of an organization can move a number of unassigned seats from one team's subscription to another team's subscription in the same organization. Nobody else can.
- Only seats nobody is using can move. The source subscription keeps at least one seat.
- Both subscriptions must be on the same plan.
- A seat belongs to one subscription; it can be assigned to a member of that team or be free.

+import { errAsync, ResultAsync } from 'neverthrow'
+import { runWithinContext, currentRequester } from '../../shared/execution-context'
+import { runWithinTransaction } from '../../shared/database-transaction'
+import { subscriptionRepository } from '../secondary-adapter/repositories'
+import { organizationRepository } from '../../organizations/secondary-adapter/repositories'
+
+export type TransferSeatsCommand = {
+  fromSubscriptionId: string
+  toSubscriptionId: string
+  seatCount: number
+}
+
+export type TransferSeatsCommandHandlerError =
+  | { kind: 'unauthenticated' }
+  | { kind: 'not-found' }
+  | { kind: 'invalid-transfer'; reason: string }
+  | { kind: 'unexpected'; cause: unknown }
+
+export function transferSeatsCommandHandler(
+  command: TransferSeatsCommand,
+): ResultAsync<void, TransferSeatsCommandHandlerError> {
+  return runWithinContext(function inContext() {
+    return runWithinTransaction({ isolationLevel: 'Serializable' }, function inTransaction() {
+      const requester = currentRequester()
+      if (!requester) return errAsync({ kind: 'unauthenticated' as const })
+
+      return ResultAsync.combine([
+        subscriptionRepository.findById(command.fromSubscriptionId),
+        subscriptionRepository.findById(command.toSubscriptionId),
+      ]).andThen(function transfer([from, to]) {
+        return organizationRepository.findById(from.organizationId).andThen(function withOrganization(organization) {
+          if (organization.ownerId !== requester.userId && !requester.isSupportAgent) {
+            return errAsync({ kind: 'invalid-transfer' as const, reason: 'only the owner can transfer seats' })
+          }
+          if (from.organizationId !== to.organizationId) {
+            return errAsync({ kind: 'invalid-transfer' as const, reason: 'different organizations' })
+          }
+          if (from.plan !== to.plan) {
+            return errAsync({ kind: 'invalid-transfer' as const, reason: 'different plans' })
+          }
+          const free = from.seats.filter(function isFree(seat) {
+            return seat.assignedMember === null
+          })
+          if (free.length < command.seatCount || from.seats.length - command.seatCount < 1) {
+            return errAsync({ kind: 'invalid-transfer' as const, reason: 'not enough free seats' })
+          }
+          const moving = free.slice(0, command.seatCount)
+          for (const seat of moving) {
+            from.seats.splice(from.seats.indexOf(seat), 1)
+            to.seats.push(seat)
+          }
+          return subscriptionRepository
+            .update(from)
+            .andThen(function updateTarget() {
+              return subscriptionRepository.update(to)
+            })
+            .map(function done() {
+              return undefined
+            })
+        })
+      })
+    })
+  })
+}
