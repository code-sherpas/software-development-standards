Task PROJ-615 «Invoicing», increment 4 of 5: **invoices paid by card are marked as paid.** Increments 1–3 (the invoice, issuing it, the payment link) are verified in production. The payment provider calls us when a payment link is paid.

Acceptance:

- When the provider sends `payment_link.paid`, the invoice behind that link is marked as paid with the moment the provider says the payment happened.
- Requests whose signature does not verify are rejected.
- The provider retries any response other than 2xx for three days, so a payment we already recorded must not fail.

Provider documentation, as pasted in the task:

> `POST` to your endpoint, header `Payflow-Signature: t=<unix seconds>,v1=<hex HMAC-SHA256 of "<t>.<raw body>">`.
> Body: `{ "id": "evt_…", "type": "payment_link.paid", "created": 1762076400, "data": { "payment_link": "plink_…", "amount": 12500, "currency": "eur", "metadata": { "invoice_id": "…" } } }`. `created` is in seconds since the Unix epoch.

+type PayflowEvent = {
+  id: string
+  type: string
+  created: number
+  data: {
+    payment_link: string
+    amount: number
+    currency: string
+    metadata: { invoice_id?: string }
+  }
+}

+function verifySignature(rawBody: string, header: string | null): boolean {
+  if (!header) return false
+  const parts = Object.fromEntries(header.split(',').map((part) => part.split('=')))
+  const expected = createHmac('sha256', process.env.PAYFLOW_WEBHOOK_SECRET!)
+    .update(`${parts.t}.${rawBody}`)
+    .digest('hex')
+  return timingSafeEqual(Buffer.from(expected), Buffer.from(parts.v1 ?? ''))
+}
+
+export async function POST(request: NextRequest) {
+  const rawBody = await request.text()
+  if (!verifySignature(rawBody, request.headers.get('payflow-signature'))) {
+    return NextResponse.json({ error: 'invalid signature' }, { status: 400 })
+  }
+
+  try {
+    const event = JSON.parse(rawBody) as PayflowEvent
+    if (event.type !== 'payment_link.paid') {
+      return NextResponse.json({ received: true })
+    }
+
+    const payment = {
+      invoiceId: event.data.metadata.invoice_id!,
+      amountCents: event.data.amount,
+      paidAt: new Date(event.created),
+    }
+
+    await prisma.payflowEvent.create({ data: { id: event.id, receivedAt: new Date() } })
+
+    const result = await markInvoiceAsPaidCommandHandler({
+      invoiceId: payment.invoiceId,
+      paidAt: payment.paidAt.toString(),
+    })
+    if (result.isErr()) {
+      console.error('payflow webhook: could not mark invoice as paid', result.error)
+      return NextResponse.json({ error: result.error.kind }, { status: 500 })
+    }
+
+    return NextResponse.json({ received: true })
+  } catch (error) {
+    console.error('payflow webhook failed', error)
+    return NextResponse.json({ error: 'internal error' }, { status: 500 })
+  }
+}
