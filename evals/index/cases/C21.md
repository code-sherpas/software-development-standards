Increment 2 of PROJ-433 (“Front desk books appointments for any of the group's clinics”).

**Increment:** the front-desk booking form gets a date-and-time field and a clinic time-zone selector, so staff in the Madrid office can book an appointment for the Lisbon or Porto clinic at the clinic's own time. The server action stores the appointment and returns it so the form can show the confirmation.

**Acceptance:**
- A receptionist picks 10:30 on 14 November and “Europe/Lisbon”; the appointment shows 10:30 at the Lisbon clinic everywhere.
- The confirmation under the form shows the booked date and time.
- Booking an appointment without a date shows an error next to the field.

+const ZONES = ['Europe/Madrid', 'Europe/Lisbon', 'Atlantic/Canary']
+
+export function AppointmentForm({ clinicianId }: { clinicianId: string }) {
+  const [error, setError] = useState<string | null>(null)
+  const [booked, setBooked] = useState<Date | null>(null)
+
+  async function handleSubmit(formData: FormData) {
+    const startsAt = formData.get('startsAt') as string
+    if (!startsAt) {
+      setError('Please choose a date and time')
+      return
+    }
+    const result = await bookAppointment({
+      clinicianId,
+      startsAt,
+      timeZone: formData.get('timeZone') as string,
+    })
+    setError(null)
+    setBooked(result.startsAt)
+  }
+
+  return (
+    <form action={handleSubmit} className="grid gap-3">
+      <input
+        type="datetime-local"
+        name="startsAt"
+        placeholder="Date and time"
+        className="border rounded px-2 py-1"
+      />
+      <select name="timeZone" defaultValue="Europe/Madrid" className="border rounded px-2 py-1">
+        {ZONES.map((zone) => (
+          <option key={zone} value={zone}>
+            {zone}
+          </option>
+        ))}
+      </select>
+      {error && <p className="text-red-500 text-xs">{error}</p>}
+      <button type="submit" className="bg-teal-500 text-white rounded px-3 py-1">

+export async function bookAppointment(input: BookAppointmentInput) {
+  const startsAt = new Date(input.startsAt)

+  const appointment = await prisma.appointment.create({

+      timeZone: input.timeZone,
