Increment 4 of PROJ-371 (“Quick-add reminders”).

**Increment:** tests only. The quick-add box already turns phrases like “next tuesday 3pm” or “tomorrow 09:30” into a reminder time with `parseQuickWhen`, a small parser in production since increment 2. Its grammar is fixed: a day part (`today`, `tomorrow`, `next <weekday>`, `<weekday>`, `in <n> days`) and an optional time part (`<h>am|pm`, `<hh>:<mm>`), nothing else. It had no tests. This increment adds them; no production code changes.

**Acceptance:** the tests cover every form of the grammar and fail if any of them breaks.

+import { describe, it, expect, vi, beforeEach } from 'vitest'
+import { Temporal } from '@js-temporal/polyfill'
+import { parseQuickWhen } from './parse-quick-when'
+import type { Clock } from '../clock'
+
+const zone = 'Europe/Madrid'
+
+let clock: Clock
+
+beforeEach(() => {
+  clock = {
+    now: vi.fn(() => Temporal.ZonedDateTime.from('2026-11-02T10:00[Europe/Madrid]')),
+  }
+})
+
+describe('parseQuickWhen', () => {
+  it('parses "next tuesday 3pm"', () => {
+    const result = parseQuickWhen('next tuesday 3pm', { clock, zone })
+
+    expect(clock.now).toHaveBeenCalledTimes(1)
+    expect(result.isOk()).toBe(true)
+  })
+
+  it('parses "tomorrow 09:30"', () => {
+    const result = parseQuickWhen('tomorrow 09:30', { clock, zone })
+
+    expect(clock.now).toHaveBeenCalledTimes(1)
+    expect(result._unsafeUnwrap().toString()).toBe('2026-11-03T09:30:00+01:00[Europe/Madrid]')
+  })
+
+  it('parses "in 3 days"', () => {
+    const result = parseQuickWhen('in 3 days', { clock, zone })
+
+    expect(clock.now).toHaveBeenCalled()
+    expect(result.isOk()).toBe(true)
+  })
+
+  it('parses "friday"', () => {
+    parseQuickWhen('friday', { clock, zone })
+
+    expect(clock.now).toHaveBeenCalledTimes(1)
+  })
+
+  it('rejects "the day after the meeting"', () => {
+    const result = parseQuickWhen('the day after the meeting', { clock, zone })
+
+    expect(result.isErr()).toBe(true)
+    expect(clock.now).not.toHaveBeenCalled()
+  })
+})
