// src/billing/application/invoice-use-cases.ts
import { ResultAsync } from 'neverthrow'
import type { InvoiceRepository } from '../domain/invoice-repository'
import { Invoice, InvoiceId } from '../domain/invoice'
import { runWithinContext } from '@/shared/execution-context'
import { runWithinTransaction } from '@/shared/transaction'

export type IssueInvoiceCommand = { invoiceId: string }
export type VoidInvoiceCommand = { invoiceId: string; reason: string }
export type FindInvoiceByIdQuery = { invoiceId: string }

export function issueInvoiceCommandHandler(command: IssueInvoiceCommand): ResultAsync<void, IssueInvoiceError> {
  return runWithinContext(() => runWithinTransaction('SERIALIZABLE', (repos) =>
    repos.invoices.findById(InvoiceId(command.invoiceId)).andThen(issue).andThen(repos.invoices.update),
  ))
}

export function voidInvoiceCommandHandler(command: VoidInvoiceCommand): ResultAsync<void, VoidInvoiceError> {
  return runWithinContext(() => runWithinTransaction('SERIALIZABLE', (repos) =>
    repos.invoices.findById(InvoiceId(command.invoiceId))
      .andThen(function voidIt(invoice) { return invoice.void(command.reason) })
      .andThen(repos.invoices.update),
  ))
}

export function findInvoiceByIdQueryHandler(query: FindInvoiceByIdQuery): ResultAsync<Invoice, FindInvoiceError> {
  return runWithinContext(() => runWithinTransaction('READ COMMITTED', (repos) =>
    repos.invoices.findById(InvoiceId(query.invoiceId)),
  ))
}

function issue(invoice: Invoice) {
  return invoice.issue()
}
