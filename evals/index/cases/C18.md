+import { errAsync, okAsync, ResultAsync } from 'neverthrow'
+import { Invoice as InvoiceRow } from '@prisma/client'
+import { runWithinContext, currentRequester } from '../../shared/execution-context'
+import { runWithinTransaction } from '../../shared/database-transaction'
+import { invoiceRepository } from '../secondary-adapter/repositories'
+import { customerRepository } from '../../customers/secondary-adapter/repositories'
+
+export type SearchInvoicesQuery = {
+  customerId: string
+  status?: 'draft' | 'issued' | 'paid' | 'void'
+  page: number
+}
+
+export type SearchInvoicesQueryHandlerError = { kind: 'forbidden' } | { kind: 'unexpected'; cause: unknown }
+
+export function searchInvoicesQueryHandler(
+  query: SearchInvoicesQuery,
+): ResultAsync<InvoiceRow[], SearchInvoicesQueryHandlerError> {
+  return runWithinContext(function inContext() {
+    return runWithinTransaction({ isolationLevel: 'ReadCommitted' }, function inTransaction() {
+      const requester = currentRequester()
+      return customerRepository
+        .findById(query.customerId)
+        .mapErr(function toUnexpected(cause) {
+          return { kind: 'unexpected' as const, cause }
+        })
+        .andThen(function withCustomer(customer) {
+          if (requester && !requester.clinicIds.includes(customer.clinicId)) {
+            return errAsync({ kind: 'forbidden' as const })
+          }
+          return invoiceRepository.findManyByCustomerId(query.customerId, query.status, query.page)
+        })
+        .map(function orEmpty(rows) {
+          return rows ?? []
+        })
+    })
+  })
+}

+import { ResultAsync } from 'neverthrow'
+import { invoiceRepository } from '../secondary-adapter/repositories'
+import { InvoiceId } from '../domain/invoice'
+
+export function checkWhetherInvoiceCanBeVoided(invoiceId: string, voidedBy: string): ResultAsync<boolean, unknown> {
+  return invoiceRepository.findById(invoiceId as InvoiceId).map(function canBeVoided(invoice) {
+    const ageInDays = Temporal.Now.instant().since(invoice.issuedAt).total({ unit: 'hours' }) / 24
+    const voidable = invoice.status === 'issued' && invoice.paidOn === undefined && ageInDays < 30
+    if (voidable) {
+      console.info(`invoice ${invoiceId} checked as voidable by ${voidedBy}`)
+    }
+    return voidable
+  })
