+      this.prisma.order.create({

+        include: { lines: true },
+      }),

+  update(command: UpdateOrderCommand): ResultAsync<OrderDto, OrderServiceError> {
+    const { order } = command
+    return this.findById(order.id).andThen((current) => {
+      if (current.status !== 'open') return errAsync({ kind: 'order-closed' as const })
+      return ResultAsync.fromPromise(
+        this.prisma.order.update({
+          where: { id: order.id },
+          data: { lines: { deleteMany: {}, create: [...order.lines] } },
+          include: { lines: true },
+        }),
+        (cause) => ({ kind: 'unexpected' as const, cause }),
+      ).map((row) => this.toDto(row))
+    })
+  }
+
+  findById(orderId: number): ResultAsync<OrderDto, OrderServiceError> {
+    return ResultAsync.fromPromise(
+      this.prisma.order.findUnique({ where: { id: orderId }, include: { lines: true } }),
+      (cause) => ({ kind: 'unexpected' as const, cause }),
+    ).andThen((row) => (row ? okAsync(this.toDto(row)) : errAsync({ kind: 'not-found' as const })))
+  }
+
+  cancel(command: CancelOrderCommand): ResultAsync<OrderDto, OrderServiceError> {
+    return this.findById(command.orderId).andThen((current) => {
+      if (current.status !== 'open') return errAsync({ kind: 'order-closed' as const })
+      return ResultAsync.fromPromise(
+        this.prisma.order.update({
+          where: { id: command.orderId },
+          data: { status: 'cancelled', cancellationReason: command.reason },
+          include: { lines: true },
+        }),
+        (cause) => ({ kind: 'unexpected' as const, cause }),
+      ).map((row) => this.toDto(row))
+    })
+  }
