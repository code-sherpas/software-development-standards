Task PROJ-472 «Quotes use the prices of the pricing service», increment 1 of 2: **a client for the pricing service.** Increment 2 switches quote creation to it; nothing calls the client yet.

Acceptance:

- Given a product SKU and a customer segment, the client returns the unit price in cents and the currency, as the pricing service has it today.
- A SKU the pricing service does not know is told apart from the pricing service being down or slow.
- The call gives up after 2 seconds.

The pricing service is another team's application in our organization. Its contract, from its README:

> `GET /v2/prices/{sku}?segment={segment}` → `200 { "sku": "…", "segment": "…", "unit_price": "12.50", "currency": "EUR", "valid_from": "2026-10-01T00:00:00.000Z" }`, `404` when the SKU is unknown, `503` during deploys.

+import * as Sentry from '@sentry/nextjs'
+import { errAsync, okAsync, ResultAsync } from 'neverthrow'
+
+export type PriceResponse = {
+  sku: string
+  segment: string
+  unit_price: string
+  currency: string
+  valid_from: string
+}
+
+export type PricingServiceError =
+  | { kind: 'unknown-sku'; sku: string }
+  | { kind: 'unavailable'; cause: unknown }
+
+const baseUrl = process.env.PRICING_SERVICE_URL ?? 'http://pricing.internal'
+
+export const pricingServiceClient = {
+  getPrice(sku: string, segment: string): ResultAsync<PriceResponse, PricingServiceError> {
+    const url = `${baseUrl}/v2/prices/${encodeURIComponent(sku)}?segment=${encodeURIComponent(segment)}`
+    return ResultAsync.fromPromise(
+      fetch(url, { signal: AbortSignal.timeout(2000) }),
+      (cause) => ({ kind: 'unavailable' as const, cause }),
+    )
+      .andThen((response) => {
+        if (response.status === 404) {
+          Sentry.captureException(new Error(`Pricing service does not know SKU ${sku}`))
+          return errAsync({ kind: 'unknown-sku' as const, sku })
+        }
+        if (!response.ok) {
+          return errAsync({ kind: 'unavailable' as const, cause: response.status })
+        }
+        return ResultAsync.fromPromise(
+          response.json() as Promise<PriceResponse>,
+          (cause) => ({ kind: 'unavailable' as const, cause }),
+        )
+      })
+      .map((body) => ({ ...body, unit_price: body.unit_price.trim() }))
+  },
+
+  async getPrices(skus: string[], segment: string): Promise<PriceResponse[]> {
+    const results = await Promise.all(skus.map((sku) => pricingServiceClient.getPrice(sku, segment)))
+    return results.filter((r) => r.isOk()).map((r) => r._unsafeUnwrap())
+  },
+}
