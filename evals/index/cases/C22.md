/build-prototype — the kanban prototype for the new “Team board” is half done and the design review is on Thursday. Can you finish it and check it works? The team will use it as the reference for how the board behaves.

What it should let people do:
- see the board with three columns (To do, Doing, Done) and move cards between them and within a column by dragging;
- open a card, edit its title, archive it;
- the toolbar has icon buttons for “new card”, “filter”, and “board settings”;
- a toast confirms when a card is archived, with an undo;
- an “Archived” screen lists archived cards and lets you restore one.

Right now the board screen and the archived screen each have their own list of cards, so archiving on the board doesn't show up in Archived. Here's what's in `prototype/team-board/`:

```tsx
// prototype/team-board/BoardScreen.tsx
import { useState } from 'react'
import { DndContext, closestCenter } from '@dnd-kit/core'
import { SortableContext, arrayMove } from '@dnd-kit/sortable'
import { Plus, Filter, Settings, X } from 'lucide-react'
import { CardTile } from './CardTile'
import { Toast } from './Toast'

const initialCards = [
  { id: 'c1', title: 'Write onboarding email', column: 'todo' },
  { id: 'c2', title: 'Fix invoice PDF footer', column: 'doing' },
  { id: 'c3', title: 'Plan Q1 offsite', column: 'todo' },
  { id: 'c4', title: 'Renew SSL certificate', column: 'done' },
]

export function BoardScreen() {
  const [cards, setCards] = useState(initialCards)
  const [toast, setToast] = useState<string | null>(null)

  return (
    <div className="p-6">
      <div className="flex gap-2 mb-4">
        <div onClick={() => alert('New card')} className="p-2 rounded hover:bg-gray-100">
          <Plus size={16} />
        </div>
        <div onClick={() => alert('Filter')} className="p-2 rounded hover:bg-gray-100">
          <Filter size={16} />
        </div>
        <div onClick={() => alert('Settings')} className="p-2 rounded hover:bg-gray-100">
          <Settings size={16} />
        </div>
      </div>

      <DndContext
        collisionDetection={closestCenter}
        onDragEnd={(event) => {
          const { active, over } = event
          if (!over || active.id === over.id) return
          setCards((items) => {
            const from = items.findIndex((c) => c.id === active.id)
            const to = items.findIndex((c) => c.id === over.id)
            return arrayMove(items, from, to)
          })
        }}
      >
        <div className="grid grid-cols-3 gap-4">
          {['todo', 'doing', 'done'].map((column) => (
            <div key={column} className="bg-gray-50 rounded p-3">
              <h2 className="text-sm text-gray-400">{column}</h2>
              <SortableContext items={cards.filter((c) => c.column === column)}>
                {cards
                  .filter((c) => c.column === column)
                  .map((card) => (
                    <CardTile
                      key={card.id}
                      card={card}
                      onArchive={() => {
                        setCards(cards.filter((c) => c.id !== card.id))
                        setToast(`Archived “${card.title}”`)
                        setTimeout(() => setToast(null), 3000)
                      }}
                    />
                  ))}
              </SortableContext>
            </div>
          ))}
        </div>
      </DndContext>

      {toast && <Toast message={toast} onClose={() => setToast(null)} icon={<X size={12} />} />}
    </div>
  )
}
```

```tsx
// prototype/team-board/ArchivedScreen.tsx
import { useState } from 'react'
import { RotateCcw } from 'lucide-react'

export function ArchivedScreen() {
  const [archived, setArchived] = useState([
    { id: 'a1', title: 'Old roadmap', column: 'done' },
    { id: 'a2', title: 'Migrate to new CI', column: 'done' },
  ])

  return (
    <ul className="p-6">
      {archived.map((card) => (
        <li key={card.id} className="flex justify-between py-2">
          {card.title}
          <button onClick={() => setArchived(archived.filter((c) => c.id !== card.id))}>
            <RotateCcw size={14} />
          </button>
        </li>
      ))}
    </ul>
  )
}
```

```tsx
// prototype/team-board/Toast.tsx
export function Toast({ message, onClose, icon }: { message: string; onClose: () => void; icon: React.ReactNode }) {
  return (
    <div className="fixed bottom-4 right-4 bg-gray-800 text-gray-300 rounded px-4 py-2 flex gap-3">
      {message}
      <span onClick={onClose}>{icon}</span>
    </div>
  )
}
```
